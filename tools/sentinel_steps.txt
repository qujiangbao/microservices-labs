3：新增网关限流配置类新建文件：flowersmall-gateway/src/main/java/cn/js/ccit/flowersmall/gateway/config/SentinelGatewayConfig.javapackage cn.js.ccit.flowersmall.gateway.config;import com.alibaba.csp.sentinel.api.gateway.GatewayFlowRule;import com.alibaba.csp.sentinel.gateway.common.rule.GatewayRuleManager;import com.alibaba.csp.sentinel.gateway.sc.callback.GatewayCallbackManager;import com.alibaba.csp.sentinel.slots.block.RuleConstant;import org.slf4j.Logger;import org.slf4j.LoggerFactory;import org.springframework.context.annotation.Configuration;import org.springframework.http.HttpStatus;import org.springframework.http.MediaType;import org.springframework.web.reactive.function.server.ServerResponse;import javax.annotation.PostConstruct;import java.util.HashMap;import java.util.HashSet;import java.util.Map;import java.util.Set;@Configurationpublic class SentinelGatewayConfig {    private static final Logger log = LoggerFactory.getLogger(SentinelGatewayConfig.class);    /** 商品服务在网关中的路由 id（= 注册到 nacos 的服务名 product） */    private static final String PRODUCT_ROUTE_ID = "product";    @PostConstruct    public void init() {        initGatewayRules();        initBlockHandler();    }    /** 网关流控规则：商品查询路由每秒最多放行 5 个请求，超过直接拒绝 */    private void initGatewayRules() {        Set<GatewayFlowRule> rules = new HashSet<>();        GatewayFlowRule productRule = new GatewayFlowRule();        productRule.setResource(PRODUCT_ROUTE_ID);        productRule.setResourceMode(GatewayRuleManager.RESOURCE_MODE_ROUTE_ID);        productRule.setGrade(RuleConstant.FLOW_GRADE_QPS);        productRule.setCount(5);        productRule.setIntervalSec(1);        rules.add(productRule);        GatewayRuleManager.loadRules(rules);        log.info("[Sentinel-Gateway] 已加载网关流控规则: {}", rules);    }    /** 自定义限流响应：触发限流时返回 429 + 友好 JSON */    private void initBlockHandler() {        GatewayCallbackManager.setBlockHandler((exchange, t) -> {            Map<String, Object> body = new HashMap<>(4);            body.put("code", HttpStatus.TOO_MANY_REQUESTS.value());            body.put("success", false);            body.put("message", "当前访问人数过多，请稍后再试（Sentinel 网关限流）");            body.put("path", exchange.getRequest().getPath().value());            log.warn("[Sentinel-Gateway] 请求被限流: path={}", exchange.getRequest().getPath().value());            return ServerResponse.status(HttpStatus.TOO_MANY_REQUESTS)                    .contentType(MediaType.APPLICATION_JSON)                    .bodyValue(body);        });    }}2.5 步骤 4：启动服务1.  启动 product /order，在创建订单后通过 Dubbo 调用 goldCoin 服务发放金币：OrderController.order()  [@GlobalTransactional]    └─(@DubboReference)→ GoldCoinService.updateGoldCoin(userName, goldCoin, orderNo)                              goldCoin 服务 @DubboService 实现：写金币明细/余额目标：对该 RPC 做流控（限制下单发金币 QPS）+ 熔断降级（goldCoin 异常率过高时熔断）+ fallback（被拦截时优雅降级，下单不报错）。3.2 实现原理引入 sentinel-apache-dubbo-adapter 后，它会通过 Dubbo 的 Filter SPI 自动织入埋点，无需修改业务代码。资源名格式为“接口全限定名:方法名(参数类型列表)”，本案例资源名为：cn.js.ccit.common.dubbo.GoldCoinService:updateGoldCoin(java.lang.String,java.lang.Integer,java.lang.String)注意：sentinel-apache-dubbo-adapter 不被 spring-cloud-alibaba BOM 管理，必须显式指定版本，且要与 BOM 管理的 Sentinel 版本一致（2021.0.1.0 → 1.8.3）。3.3 步骤 1：order/pom.xml 增加依赖在 dubbo 依赖之后、security 依赖之前加入：<!-- Sentinel：对 order->goldCoin 的 Dubbo 调用做流控 + 熔断降级 --><dependency>    <groupId>com.alibaba.cloud</groupId>    <artifactId>spring-cloud-starter-alibaba-sentinel</artifactId></dependency><!-- Sentinel 对 Apache Dubbo(2.7.x) 的适配：通过 Dubbo Filter SPI com.alibaba.csp.sentinel.adapter.dubbo.config.DubboAdapterGlobalConfig;import com.alibaba.csp.sentinel.slots.block.RuleConstant;import com.alibaba.csp.sentinel.slots.block.degrade.CircuitBreakerStrategy;import com.alibaba.csp.sentinel.slots.block.degrade.DegradeRule;import com.alibaba.csp.sentinel.slots.block.degrade.DegradeRuleManager;import com.alibaba.csp.sentinel.slots.block.flow.FlowRule;import com.alibaba.csp.sentinel.slots.block.flow.FlowRuleManager;import org.apache.dubbo.rpc.AsyncRpcResult;import org.slf4j.Logger;import org.slf4j.LoggerFactory;import org.springframework.context.annotation.Configuration;import javax.annotation.PostConstruct;import java.util.Collections;@Configurationpublic class SentinelDubboConfig {    private static final Logger log = LoggerFactory.getLogger(SentinelDubboConfig.class);    private static final String GOLD_COIN_RESOURCE =            "cn.js.ccit.common.dubbo.GoldCoinService:updateGoldCoin(java.lang.String,java.lang.Integer,java.lang.String)";    @PostConstruct    public void init() {        initFlowRules();        initDegradeRules();        initConsumerFallback();    }    /** 流控：下单发金币 QPS 超过 2 直接拒绝（演示阈值） */    private void initFlowRules() {        FlowRule rule = new FlowRule();        rule.setResource(GOLD_COIN_RESOURCE);        rule.setGrade(RuleConstant.FLOW_GRADE_QPS);        rule.setCount(2);        rule.setLimitApp("default");        FlowRuleManager.loadRules(Collections.singletonList(rule));        log.info("[Sentinel-Dubbo] 已加载流控规则: {}", rule);    }    /** 熔断降级：10s 内请求数>=5 且异常比例>50% 时熔断 10s */    private void initDegradeRules() {        DegradeRule rule = new DegradeRule();        rule.setResource(GOLD_COIN_RESOURCE);        rule.setGrade(CircuitBreakerStrategy.ERROR_RATIO.getType());        rule.setCount(0.5);        rule.setMinRequestAmount(5);        rule.setStatIntervalMs(10000);        rule.setTimeWindow(10);        DegradeRuleManager.loadRules(Collections.singletonList(rule));        log.info("[Sentinel-Dubbo] 已加载熔断降级规则: resource={}", GOLD_COIN_RESOURCE);    }    /** 消费端 fallback：被限流/熔断时优雅降级（void 返回 null），下单不报错 */    private void initConsumerFallback() {        DubboAdapterGlobalConfig.setConsumerFallback((invoker, invocation, ex) -> {            log.warn("[Sentinel-Dubbo] 下单发金币被拦截(限流/熔断): method={}, rule={}",                    invocation.getMethodName(),                    ex.getRule() == null ? ex.getClass().getSimpleName() : ex.getRule());            return AsyncRpcResult.newDefaultAsyncResult((Object) null, invocation);        });    }}3.8 步骤 6：新增提供端配置类（goldCoin）新建文件：flowersmall-goldCoin/src/main/java/cn/js/ccit/flowersmall/goldCoin/config/SentinelDubboProviderConfig.javapackage org.slf4j.LoggerFactory;import org.springframework.context.annotation.Configuration;import javax.annotation.PostConstruct;import java.util.Collections;@Configurationpublic class SentinelDubboProviderConfig {    private static final Logger log = LoggerFactory.getLogger(SentinelDubboProviderConfig.class);    private static final String GOLD_COIN_RESOURCE =            "cn.js.ccit.common.dubbo.GoldCoinService:updateGoldCoin(java.lang.String,java.lang.Integer,java.lang.String)";    @PostConstruct    public void init() {        initFlowRules();                 invocation.getMethodName(),                    ex.getRule() == null ? ex.getClass().getSimpleName() : ex.getRule());            return AsyncRpcResult.newDefaultAsyncResult((Object) null, invocation);        });    }}3.9 步骤 7：启动服务1.  依次启动 goldCoin（8003）、order（8002）、user、gateway（9000）。2.  order 启动日志出现 [Sentinel-Dubbo] 已加载流控规则/熔断降级规则；goldCoin 启动日志出现 [Sentinel-Dubbo-Provider] fallback 优雅降级：返回 null 使下单继续、事务提交（仅漏发金币，已告警）。•  若要求“宁可下单失败也不漏发金币”：把 fallback 改为 return AsyncRpcResult.newDefaultAsyncResult(ex.toRuntimeException(), invocation); 异常上抛会触发 order() 上 @GlobalTransactional 前缀。•  Dubbo 规则不生效：核对资源名严格等于 接口全限定名:方法名(参数类型)，参数类型用全限定名且以英文逗号相连、无空格。•  依赖冲突/找不到类：sentinel-apache-dubbo-adapter 版本必须为 1.8.3，与 spring-cloud-alibaba 2021.0.1.0 的 Sentinel 版本对齐。•  下单直接报错而非降级：检查是否误用了抛异常的 fallback；优雅降级应返回 (Object) null。
